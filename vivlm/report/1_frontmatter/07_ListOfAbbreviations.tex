\chapter*{DANH MỤC TỪ VIẾT TẮT}
\addcontentsline{toc}{chapter}{DANH MỤC TỪ VIẾT TẮT}

\begin{abbrv}
	\item \textbf{BLEU} \quad Bilingual Evaluation Understudy
	\item \textbf{BPC} \quad Bits Per Character (số bit trên mỗi ký tự)
	\item \textbf{BPE} \quad Byte-Pair Encoding
	\item \textbf{CIDEr} \quad Consensus-based Image Description Evaluation
	\item \textbf{FFN} \quad Feed-Forward Network
	\item \textbf{LLM} \quad Large Language Model (mô hình ngôn ngữ lớn)
	\item \textbf{LM} \quad Language Model (mô hình ngôn ngữ)
	\item \textbf{PPL} \quad Perplexity (độ hỗn loạn)
	\item \textbf{RL} \quad Reinforcement Learning (học tăng cường)
	\item \textbf{RoPE} \quad Rotary Position Embedding
	\item \textbf{RMSNorm} \quad Root Mean Square Layer Normalization
	\item \textbf{SCST} \quad Self-Critical Sequence Training
	\item \textbf{SFT} \quad Supervised Fine-Tuning (tinh chỉnh có giám sát)
	\item \textbf{VLM} \quad Vision--Language Model (mô hình ngôn ngữ--thị giác)
	\item \textbf{VQA} \quad Visual Question Answering (hỏi--đáp trên ảnh)
\end{abbrv}
